composition-kubernetes-namespace.yaml

apiVersion: apiextensions.crossplane.io/v1
kind: CompositeResourceDefinition
metadata:
  name: xkubernetesnamespaces.kubernetes.crossplane.io
spec:
  group: kubernetes.crossplane.io
  names:
    kind: XKubernetesNamespace
    plural: xkubernetesnamespaces
    shortNames:
      - xkns
  claimNames:
    kind: KubernetesNamespace
    plural: kubernetesnamespaces
    shortNames:
      - kns
  versions:
    - name: v1alpha1
      served: true
      referenceable: true
      schema:
        openAPIV3Schema:
          type: object
          properties:
            spec:
              type: object
              properties:
                id:
                  type: string
                  description: "El nombre del Namespace de Kubernetes a crear."
                  maxLength: 63
                  pattern: "^[a-z0-9]([-a-z0-9]*[a-z0-9])?$"
              required:
                - id
---
apiVersion: apiextensions.crossplane.io/v1
kind: Composition
metadata:
  name: xkubernetesnamespaces.kubernetes.crossplane.io
  labels:
    crossplane.io/xrd: xkubernetesnamespaces.kubernetes.crossplane.io
spec:
  compositeTypeRef:
    apiVersion: kubernetes.crossplane.io/v1alpha1
    kind: XKubernetesNamespace
  resources:
    - name: namespace
      base:
        apiVersion: v1
        kind: Namespace
        metadata:
          labels: {}
      patches:
        - fromFieldPath: spec.id
          toFieldPath: metadata.name
          type: FromCompositeFieldPath
        - fromFieldPath: metadata.labels
          toFieldPath: metadata.labels
          type: FromCompositeFieldPath
          policy:
            fromFieldPath: Optional
