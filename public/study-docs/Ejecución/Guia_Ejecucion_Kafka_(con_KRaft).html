kafka/entrypoint.sh

#!/bin/bash
set -e

# Path to Kafka installation
KAFKA_HOME="/opt/kafka"
KAFKA_BIN="${KAFKA_HOME}/bin"
SERVER_PROPERTIES="${KAFKA_HOME}/config/server.properties"
KAFKA_DATA_DIR="/bitnami/kafka/data" # Keep consistent with original volume

# Ensure the data directory exists
mkdir -p "${KAFKA_DATA_DIR}"

# Copy default Kraft server.properties
cp "${KAFKA_HOME}/config/kraft/server.properties" "${SERVER_PROPERTIES}"

# Apply configurations from environment variables
# KAFKA_CFG_NODE_ID -> node.id
# KAFKA_CFG_PROCESS_ROLES -> process.roles
# etc.
for VAR in $(env | grep "^KAFKA_CFG_" | sort -r); do
    KEY=$(echo "$VAR" | sed -r "s/KAFKA_CFG_(.*)=.*/\1/g" | tr '[:upper:]' '[:lower:]' | sed -r "s/_/./g")
    VALUE=$(echo "$VAR" | sed -r "s/KAFKA_CFG_.*=(.*)/\1/g")
    # Replace or append
    if grep -q "^${KEY}=" "${SERVER_PROPERTIES}"; then
        sed -i "s|^${KEY}=.*|${KEY}=${VALUE}|g" "${SERVER_PROPERTIES}"
    else
        echo "${KEY}=${VALUE}" >> "${SERVER_PROPERTIES}"
    fi
done

# Ensure log.dirs is set and points to the volume
echo "log.dirs=${KAFKA_DATA_DIR}/kraft-combined-logs" >> "${SERVER_PROPERTIES}"

echo "Final server.properties:"
cat "${SERVER_PROPERTIES}"

# Check if storage is formatted. If not, format it.
# Look for meta.properties in the data directory.
if [ ! -f "${KAFKA_DATA_DIR}/meta.properties" ]; then
    echo "Kafka storage not formatted. Generating a new cluster ID and formatting storage..."
    CLUSTER_ID=$(kafka-storage.sh random-uuid)
    echo "Generated cluster ID: ${CLUSTER_ID}"
    kafka-storage.sh format -t "${CLUSTER_ID}" -c "${SERVER_PROPERTIES}"
else
    echo "Kafka storage already formatted. Skipping format step."
fi

# Execute the original command (kafka-server-start.sh)
exec "$@"
