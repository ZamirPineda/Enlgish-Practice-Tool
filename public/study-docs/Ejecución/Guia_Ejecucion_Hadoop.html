Dockerfile

FROM ubuntu:22.04

# Install necessary packages (Java, SSH, Rsync, Curl, Wget, Net-tools, IP-Utils)
RUN apt-get update && \
    apt-get install -y openjdk-11-jdk-headless ssh rsync curl wget net-tools iputils-ping && \
    apt-get clean && \
    rm -rf /var/lib/apt/lists/*

# Set JAVA_HOME environment variable
ENV JAVA_HOME=/usr/lib/jvm/java-11-openjdk-amd64
ENV PATH=$PATH:$JAVA_HOME/bin

# Define Hadoop version as an argument for easier updates
ARG HADOOP_VERSION=3.3.6
ARG HADOOP_URL=https://dlcdn.apache.org/hadoop/common/hadoop-${HADOOP_VERSION}/hadoop-${HADOOP_VERSION}.tar.gz

# Download and extract Hadoop
RUN wget ${HADOOP_URL} -O /tmp/hadoop.tar.gz && \
    tar -xzvf /tmp/hadoop.tar.gz -C /opt && \
    rm /tmp/hadoop.tar.gz

# Set Hadoop environment variables
ENV HADOOP_HOME=/opt/hadoop-${HADOOP_VERSION}
ENV HADOOP_CONF_DIR=$HADOOP_HOME/etc/hadoop
ENV PATH=$PATH:$HADOOP_HOME/bin:$HADOOP_HOME/sbin

# Create directories for HDFS and YARN data, and temporary directories
# These directories are critical for HDFS NameNode and DataNode operations and YARN logs/timeline. 
# Permissions are set to allow Hadoop to write to them.
RUN mkdir -p $HADOOP_HOME/data/dfs/namenode && \
           mkdir -p $HADOOP_HOME/data/dfs/datanode && \
           mkdir -p $HADOOP_HOME/data/yarn/timeline && \
           mkdir -p $HADOOP_HOME/tmp && \
           chmod -R 777 $HADOOP_HOME/data && \
           chmod -R 777 $HADOOP_HOME/tmp

# Set up SSH for inter-container communication (optional for basic setup, but good practice for full cluster management scripts)
# For a single container setup with direct service commands, SSH might not be strictly necessary for starting services, 
# but it's a common Hadoop setup step.
RUN ssh-keygen -A && \
    ssh-keygen -t rsa -P '' -f ~/.ssh/id_rsa && \
    cat ~/.ssh/id_rsa.pub >> ~/.ssh/authorized_keys && \
    chmod 0600 ~/.ssh/authorized_keys

# Expose default Hadoop ports:
# 9870: NameNode UI (for HDFS)
# 8088: ResourceManager UI (for YARN)
# 19888: MapReduce JobHistory Server UI
# 8042: NodeManager UI (for YARN)
EXPOSE 9870 8088 19888 8042

# Set the working directory to Hadoop home
WORKDIR $HADOOP_HOME

# Default command to keep the container running if no other command is specified by docker-compose
CMD ["/bin/bash"]

docker-compose.yml

version: '3.8'

services:
  namenode:
    build:
      context: .
      dockerfile: Dockerfile
    hostname: namenode
    # Map ports for external access to UIs
    ports:
      - "9870:9870" # NameNode UI
      - "8088:8088" # ResourceManager UI
      - "19888:19888" # MapReduce JobHistory Server
    # Mount configuration files and data volumes
    volumes:
      - ./hadoop-config:/opt/hadoop-3.3.6/etc/hadoop # Mount custom Hadoop configuration
      - namenode_data:/opt/hadoop-3.3.6/data/dfs/namenode # Persistent storage for NameNode metadata
    environment:
      - HADOOP_VERSION=3.3.6 # Ensure consistency with Dockerfile
    # Load additional environment variables from hadoop-env.sh
    env_file:
      - ./hadoop-config/hadoop-env.sh
    # Command to format the NameNode (if not already formatted), then start NameNode, ResourceManager, and HistoryServer
    # The `&& /bin/bash` keeps the container running after service startup
    command: ["/bin/bash", "-c", "hdfs namenode -format -force -nonInteractive && hdfs --daemon start namenode && yarn --daemon start resourcemanager && mapred --daemon start historyserver && /bin/bash"]
    # Healthcheck to ensure NameNode UI is reachable before dependent services start
    healthcheck:
      test: ["CMD-SHELL", "curl -f http://localhost:9870 || exit 1"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - hadoop-net

  datanode:
    build:
      context: .
      dockerfile: Dockerfile
    hostname: datanode1
    # Ensure NameNode is healthy before starting DataNode
    depends_on:
      namenode:
        condition: service_healthy
    # Mount configuration files and data volumes
    volumes:
      - ./hadoop-config:/opt/hadoop-3.3.6/etc/hadoop # Mount custom Hadoop configuration
      - datanode_data:/opt/hadoop-3.3.6/data/dfs/datanode # Persistent storage for DataNode blocks
    environment:
      - HADOOP_VERSION=3.3.6 # Ensure consistency with Dockerfile
    # Load additional environment variables from hadoop-env.sh
    env_file:
      - ./hadoop-config/hadoop-env.sh
    # Command to start DataNode and NodeManager
    command: ["/bin/bash", "-c", "hdfs --daemon start datanode && yarn --daemon start nodemanager && /bin/bash"]
    # Expose NodeManager UI (for the single DataNode)
    ports:
      - "8042:8042" # NodeManager UI
    networks:
      - hadoop-net

# Define a custom bridge network for inter-container communication
networks:
  hadoop-net:
    driver: bridge

# Define Docker volumes for persistent storage of HDFS data
volumes:
  namenode_data:
  datanode_data:
