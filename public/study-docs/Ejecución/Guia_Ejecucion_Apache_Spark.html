Dockerfile

FROM openjdk:11-jre-slim

# Descargar y configurar Apache Spark
ARG SPARK_VERSION=3.5.1
ARG HADOOP_VERSION=3
ENV SPARK_HOME /spark

# Instalar wget y bash (necesario para algunas distribuciones slim y para ejecutar scripts)
RUN apt-get update && apt-get install -y wget bash && rm -rf /var/lib/apt/lists/*

# Descargar, extraer y mover Spark a /spark
RUN wget -qO- https://archive.apache.org/dist/spark/spark-${SPARK_VERSION}/spark-${SPARK_VERSION}-bin-hadoop${HADOOP_VERSION}.tgz | tar xvz -C /opt/
RUN mv /opt/spark-${SPARK_VERSION}-bin-hadoop${HADOOP_VERSION} ${SPARK_HOME}

# Establecer variables de entorno para que Spark estÃ© en el PATH
ENV PATH="${SPARK_HOME}/bin:${SPARK_HOME}/sbin:${PATH}"

# Exponer puertos que Spark usa (Master UI y RPC)
EXPOSE 8080 7077

docker-compose.yml

version: '3.8'

services:
  spark-master:
    build:
      context: .
      dockerfile: Dockerfile
    container_name: spark-master
    hostname: spark-master
    ports:
      - "8080:8080" # Spark Master UI
      - "7077:7077" # Spark Master internal communication
    environment:
      - SPARK_MASTER_HOST=spark-master
      - SPARK_MASTER_PORT=7077
      - SPARK_MASTER_WEBUI_PORT=8080
    command: bin/spark-class org.apache.spark.deploy.master.Master -p 7077 --webui-port 8080

  spark-worker:
    build:
      context: .
      dockerfile: Dockerfile
    container_name: spark-worker
    hostname: spark-worker
    depends_on:
      - spark-master
    environment:
      - SPARK_MASTER_URL=spark://spark-master:7077
      - SPARK_WORKER_WEBUI_PORT=8081
    ports:
      - "8081:8081" # Spark Worker UI (primer trabajador)
    command: bin/spark-class org.apache.spark.deploy.worker.Worker spark://spark-master:7077 --webui-port 8081

  spark-worker-2:
    build:
      context: .
      dockerfile: Dockerfile
    container_name: spark-worker-2
    hostname: spark-worker-2
    depends_on:
      - spark-master
    environment:
      - SPARK_MASTER_URL=spark://spark-master:7077
      - SPARK_WORKER_WEBUI_PORT=8082
    ports:
      - "8082:8082" # Spark Worker UI (segundo trabajador)
    command: bin/spark-class org.apache.spark.deploy.worker.Worker spark://spark-master:7077 --webui-port 8082

  spark-history:
    build:
      context: .
      dockerfile: Dockerfile
    container_name: spark-history
    hostname: spark-history
    depends_on:
      - spark-master
    ports:
      - "18080:18080" # Spark History Server UI
    environment:
      # Opciones del History Server para indicar dÃ³nde buscar los logs de eventos
      - SPARK_HISTORY_OPTS="-Dspark.history.fs.logDirectory=/tmp/spark-events -Dspark.history.fs.update.interval=10s"
    volumes:
      - spark-events:/tmp/spark-events # Montar volumen para persistir los logs de eventos
    command: bin/spark-class org.apache.spark.deploy.history.HistoryServer

volumes:
  spark-events: # DefiniciÃ³n del volumen para los logs del historial
